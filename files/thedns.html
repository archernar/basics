#!/usr/bin/bash
# This script performs DNS lookups for a specified domain.
# It uses the 'dig' command, which is a powerful tool for querying DNS servers.

# Function to perform DNS lookup and handle errors
lookup_record() {
  local domain="$1"
  local record_type="$2"

  # Use dig to query the DNS record, include error handling
  if ! dig "$domain" "$record_type" +short >/dev/null 2>&1; then
    echo "Error: Could not retrieve $record_type records for $domain"
    return 1 # Return a non-zero exit code to indicate failure
  fi

  # added the check to make sure the output is printed only if the command was successful
  dig "$domain" "$record_type" +short
}

# Check if a domain name is provided as a command-line argument
if [ -z "$1" ]; then
  echo "Usage: $0 "
  echo "  Example: $0 google.com"
  exit 1
fi

# Store the domain name from the first command-line argument
domain="$1"

# Perform lookups for various DNS record types
echo "--------------------------------------------------------"
echo "DNS Records for: $domain"
echo "--------------------------------------------------------"

echo
echo "SOA Records:"
lookup_record "$domain" SOA 
echo

echo "A Records:"
lookup_record "$domain" A
echo

echo "AAAA Records (IPv6):"
lookup_record "$domain" AAAA
echo

echo "MX Records (Mail Exchange):"
lookup_record "$domain" MX
echo

echo "SPF Records (SPF):"
lookup_record "$domain" SPF
echo

echo "CNAME Records (Canonical Name):"
lookup_record "$domain" CNAME
echo

echo "TXT Records (Text):"
lookup_record "$domain" TXT
echo

echo "PTR Records (PTR):"
lookup_record "$domain" PTR
echo

echo "NS Records (Name Server):"
lookup_record "$domain" NS
echo

# Script execution completed.  The exit code of the *last* command
# executed (dig, in this case) will be the script's exit code.
# If all dig commands were successful, the exit code will be 0.
exit 0
